import chess, random, subprocess
roots=[chess.STARTING_FEN,"r3k2r/p1ppqpb1/bn2pnp1/3PN3/1p2P3/2N2Q1p/PPPBBPPP/R3K2R w KQkq - 0 1","8/PPPk4/8/8/8/8/4Kppp/8 w - - 0 1","8/2p5/3p4/KP5r/1R3p1k/8/4P1P1/8 w - - 0 1"]
cases=[]
for j,fen in enumerate(roots):
 for seed in range(5):
  rng=random.Random(5000+j*100+seed);b=chess.Board(fen)
  for ply in range(100):
   moves=list(b.legal_moves)
   cases.append((b.fen(en_passant="fen"), sorted(m.uci() for m in moves), sorted(m.uci() for m in moves if b.is_capture(m)),b.fen(en_passant="legal")))
   if not moves: break
   b.push(rng.choice(moves))
p=subprocess.run([r"C:\Users\macur\AppData\Local\Temp\rarog-board-audit-20260905\board-audit-bridge.exe"],input="\n".join(c[0] for c in cases)+"\n",text=True,capture_output=True,timeout=90)
assert p.returncode==0,(p.returncode,p.stderr[-3000:])
lines=p.stdout.splitlines();assert len(lines)==len(cases),(len(lines),len(cases))
for row,(fen,moves,caps,canonical) in zip(lines,cases):
 a,b,c=row.split("|")
 assert ([] if not a else a.split(","))==moves,("moves",fen,a,moves)
 assert ([] if not b else b.split(","))==caps,("captures",fen,b,caps)
 assert c==canonical,("FEN",fen,c,canonical)
print(p.stderr.strip());print("PASS independent python-chess positions =",len(cases),"legal moves and hinted make/unmakes =",sum(len(c[1]) for c in cases),"captures =",sum(len(c[2]) for c in cases))
