use rarog::board::{Board, Bitboard, Square, ATTACKS};
use std::io::{self,BufRead};
fn ray(s:u8,occ:u64,dirs:&[(i8,i8)])->u64{let mut out=0;for &(df,dr) in dirs{let(mut f,mut r)=((s%8)as i8+df,(s/8)as i8+dr);while (0..8).contains(&f)&&(0..8).contains(&r){let bit=1u64<<((r*8+f)as u8);out|=bit;if occ&bit!=0{break;}f+=df;r+=dr;}}out}
fn sliders(){let mut n=0u64;for rook in [false,true]{let dirs:&[(i8,i8)]=if rook{&[(1,0),(-1,0),(0,1),(0,-1)]}else{&[(1,1),(-1,1),(1,-1),(-1,-1)]};for s in 0..64u8{let mut mask=0u64;for &(df,dr) in dirs{let(mut f,mut r)=((s%8)as i8+df,(s/8)as i8+dr);while (0..8).contains(&(f+df))&&(0..8).contains(&(r+dr)){mask|=1u64<<((r*8+f)as u8);f+=df;r+=dr;}}let mut sub=0u64;loop{let actual=if rook{ATTACKS.rook(Square(s),Bitboard(sub))}else{ATTACKS.bishop(Square(s),Bitboard(sub))};assert_eq!(actual.0,ray(s,sub,dirs));n+=1;sub=sub.wrapping_sub(mask)&mask;if sub==0{break;}}}}eprintln!("SLIDER_CASES={n}");}
fn main(){sliders();for line in io::stdin().lock().lines(){let line=line.unwrap();let mut b=Board::from_fen(&line).unwrap();let before=b.to_fen();let h=b.hash;let moves=b.generate_legal_movelist();let ci=b.check_info();let mut names=vec![];for &m in &moves{let hint=b.gives_check_with(m,&ci);b.make_move_with_check(m,hint);b.check_consistency().unwrap();b.unmake_move(m);b.check_consistency().unwrap();assert_eq!(h,b.hash);assert_eq!(before,b.to_fen());names.push(m.to_string());}names.sort();let mut caps:Vec<String>=b.generate_legal_captures().iter().map(|m|m.to_string()).collect();caps.sort();if !b.is_in_check(){b.make_null_move();b.check_consistency().unwrap();b.unmake_null_move();b.check_consistency().unwrap();assert_eq!(before,b.to_fen());assert_eq!(h,b.hash);}println!("{}|{}|{}",names.join(","),caps.join(","),b.to_fen());}}
